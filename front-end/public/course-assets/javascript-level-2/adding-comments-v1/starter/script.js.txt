function normalizeDocument(value) {

function seedDocument() {

}
function addTopic(state, title, url, imageUrl, id) {
	if (state.posts.length >= 30 || state.posts.some((post) => post.id === id))
		throw new StoreProblem("Topic limit or repeated ID.");
	const order =
		state.posts.reduce((maximum, post) => Math.max(maximum, post.order), 0) + 1;
	if (order > 1000000000) throw new StoreProblem("Post order limit reached.");
	const post = {
		id: recordId(id),
		title: textField(title, 100),
		url: reviewedUrl(url),
		imageUrl: reviewedUrl(imageUrl, true),
		order,
	};
	post.comments = [];
	state.posts.push(post);
	return true;
}

function renderTopics(state) {
	const output = document.querySelector("#topics");
	output.replaceChildren();
	const posts = [...state.posts].sort(
		(a, b) => b.order - a.order || a.id.localeCompare(b.id),
	);
	for (const post of posts) {
		const article = document.createElement("article"),
			heading = document.createElement("h2"),
			link = document.createElement("a");
		article.dataset.postId = post.id;
		link.href = post.url;
		link.target = "_blank";
		link.rel = "noopener noreferrer";
		link.textContent = post.title;
		heading.append(link);
		const image = document.createElement("img"),
			fallback = document.createElement("p");
		image.width = 240;
		image.alt = post.imageUrl ? "Image for " + post.title : "";
		fallback.className = "image-status";
		fallback.textContent = post.imageUrl
			? "Image unavailable."
			: "No image address supplied.";
		fallback.hidden = !!post.imageUrl;
		image.onerror = () => {
			image.hidden = true;
			fallback.hidden = false;
		};
		image.src = post.imageUrl || suppliedIllustration;
		article.append(heading, image, fallback);
		{
			const label = document.createElement("label"),
				input = document.createElement("input"),
				button = document.createElement("button"),
				form = document.createElement("form"),
				list = document.createElement("ol");
			form.className = "comment-form";
			form.dataset.postId = post.id;
			input.id = "comment-" + post.id;
			input.maxLength = 400;
			input.required = true;
			label.htmlFor = input.id;
			label.textContent = "Comment on " + post.title;
			button.type = "submit";
			button.textContent = "Add comment";
			form.append(label, input, button);
			form.addEventListener("submit", (event) => {
				event.preventDefault();
				if (
					perform(
						(state) => addComment(state, post.id, input.value, newRecordId()),
						"Comment saved.",
					)
				) {
					document
						.querySelector('form[data-post-id="' + post.id + '"] input')
						?.focus();
				}
			});
			list.className = "comments";
			for (const comment of post.comments) {
				const item = document.createElement("li");
				item.dataset.commentId = comment.id;
				item.textContent = comment.text;
				list.append(item);
			}
			article.append(list, form);
		}
		output.append(article);
	}
	if (!posts.length) {
		const empty = document.createElement("p");
		empty.textContent = "No topics.";
		output.append(empty);
	}
}
const store = openDocument(
	() => localStorage,
	"classes-course:" + PROJECT_NAME + ":document",
	normalizeDocument,
	seedDocument,
);
function redraw() {
	try {
		renderTopics(store.state);
		return null;
	} catch (error) {
		return error;
	}
}
function perform(mutate, message) {
	try {
		const changed = commitDocument(store, mutate, normalizeDocument),
			renderError = redraw();
		setStatus(
			renderError
				? (changed ? "Saved locally. " : "") + renderError.message
				: changed
					? message
					: "The selected topic is not present.",
			renderError ? "error" : "success",
		);
		return changed;
	} catch (error) {
		redraw();
		setStatus(error.message, "error");
		return false;
	}
}
document.querySelector("#topic-form").addEventListener("submit", (event) => {
	event.preventDefault();
	const title = document.querySelector("#title"),
		url = document.querySelector("#url"),
		image = document.querySelector("#image");
	if (
		perform(
			(state) =>
				addTopic(state, title.value, url.value, image.value, newRecordId()),
			"Topic saved.",
		)
	) {
		title.value = "";
		url.value = "";
		image.value = "";
		title.focus();
	}
});
document.querySelector("#reset").addEventListener("click", () => {
	if (!confirm("Replace this project's local data with the supplied example?"))
		return;
	perform((state) => {
		const seed = seedDocument();
		state.schemaVersion = seed.schemaVersion;
		state.posts = seed.posts;
		return true;
	}, "Example restored.");
	document.querySelector("#title").focus();
});
