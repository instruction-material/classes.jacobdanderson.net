PRAGMA foreign_keys = ON;
CREATE TABLE bookshelves (
  id INTEGER PRIMARY KEY,
  label TEXT NOT NULL UNIQUE
);
CREATE TABLE books (
  id INTEGER PRIMARY KEY,
  title TEXT NOT NULL,
  shelf_id INTEGER NOT NULL REFERENCES bookshelves(id)
);
